\h{1 · Schutzziele \& Bedrohungen (STRIDE)}

\tabstart
\begin{tabularx}{\linewidth}{K{0.34\linewidth}L}
\THH{Schutzziel} & \THH{Bedrohung / Gegenmaßnahme}\\
\textbf{Authentizität}         & Spoofing $\to$ Signaturen, Zertifikate\\
\textbf{Integrität}            & Tampering $\to$ MAC, AEAD, Hash\\
\textbf{Nicht-Abstreitbarkeit} & Repudiation $\to$ Digitale Signatur\\
\textbf{Vertraulichkeit}       & Information Disclosure $\to$ Verschlüsselung\\
\textbf{Verfügbarkeit}         & Denial of Service $\to$ Rate-Limiting\\
\textbf{Autorisierung}         & Privilege Escalation $\to$ ACLs\\
\end{tabularx}
\tabend

\begin{itemize}
\item \textbf{Kerckhoffs:} System sicher auch wenn alles außer Schlüssel bekannt
  $\to$ kein Security by Obscurity
\item \textbf{Bedrohung beseitigen} $>$ Schwachstelle absichern.
  Risiko = Bedrohung $\times$ Schwachstelle. Sicherheit = Kosten-Nutzen
\item \textbf{Angreifermodell:} beschreibt Fähigkeiten des Angreifers
  (= erlaubte Einsatzumgebung). Strenger = stärker
\end{itemize}

\h{2 · Kryptographische Grundbegriffe}
\begin{itemize}
\item \textbf{Chiffre} $\Pi$=(E,D): $D(k,E(k,m))=m$.
  \textbf{K}~Keyspace, \textbf{M}~Messagespace, \textbf{C}~Cipherspace
\item \textbf{Schema} $\Pi$=(KGen,E,D): KGen randomisiert $\to k\in K$;
  E darf randomisiert sein; D deterministisch.\\
  \cd{r$\leftarrow$A(x)} randomisiert; \cd{y:=A(x)} deterministisch
\item \textbf{Klassische Chiffren:} Caesar K=$\{3\}$, ROT K=$\{0..25\}$.
  Problem: \textbf{Häufigkeitsanalyse} -- PT-Statistik im CT sichtbar.
\end{itemize}

\h{3 · Perfect Secrecy \& One-Time-Pad}
\begin{itemize}
\item \textbf{XOR:} $a\xor a=0$, $a\xor 0=a$, $a\xor b\xor b=a$,
  selbstinvers, assoziativ, kommutativ
\item \textbf{OTP:} $K=M=C=\mathbb{B}^n$, $k$ uniform.
  $E(k,m)=m\xor k$,\; $D(k,c)=c\xor k$
\end{itemize}
\begin{ibox}
\textbf{Perfect Secrecy:} $\Pr[M=m\mid C=c]=\Pr[M=m]$ für alle $m,c$
$\to$ CT verrät \textbf{nichts} über PT\\
\textbf{Orakel-Spiel:} kein poly-Zeit-Angreifer gewinnt Unterscheidungsspiel
mit $\Pr>\tfrac{1}{2}$ (Raten $=\tfrac{1}{2}$)\\
\textbf{Schranke:} P.S.\ $\Rightarrow |K|\ge|M|$
(bei $|K|<|M|$: $\exists m$ mit $\Pr[C=c\mid M=m]=0$)\\
\textbf{Shannon-Satz} ($|K|{=}|M|{=}|C|$): P.S.\ $\Leftrightarrow$
(i) $k$ uniform \textbf{und} (ii) zu $(m,c)$ genau ein $k$ mit $E_k(m)=c$
\end{ibox}
\begin{itemize}
\item \textbf{Bayes:}
  $\Pr[M{=}m|C{=}c]=\frac{\Pr[C{=}c|M{=}m]\cdot\Pr[M{=}m]}{\Pr[C{=}c]}
  \Rightarrow \Pr[E_k(m){=}c]$ unabhängig von $m$
\item \warn{OTP:} Schlüssel so lang wie Nachricht; echter Zufall; nie zweimal!
\end{itemize}

\h{4 · Block- \& Streamchiffren}
\begin{itemize}
\item \textbf{AES} (Blockchiffre): Block immer 128\,Bit;
  Schlüssel 128/192/256\,Bit. NIST 2001
\item \textbf{ChaCha20} (Streamchiffre): Keystream aus (Key+Nonce) $\xor$ PT.
  TLS~1.3 Standard. RC4 = \warn{unsicher}
\end{itemize}
\begin{wbox}
\textbf{Nonce-Reuse:} Gleiche (Nonce,Key) $\Rightarrow$ gleicher Keystream
$\Rightarrow c\xor c'=m\xor m'$ (statistisch analysierbar).
Verfahren \textbf{komplett gebrochen!}\\
\end{wbox}

\h{5 · Betriebsmodi ($>$1 Block)}
\begin{itemize}
\item \textbf{PKCS\#7 Padding:} fehlende Bytes mit Wert=Anzahl füllen (1..16).
  Voller Extrablock wenn $|m|\bmod n=0$. Letztes Byte = Anzahl. Injektiv.
\end{itemize}

\tabstart
\begin{tabularx}{\linewidth}{K{0.13\linewidth}L}
\THH{Modus} & \THH{Verschlüsselung / Eigenschaften}\\
\textbf{ECB} & $c_i=E(k,m_i)$. Pad: ja.
  \warn{NIE! Gleiche Blöcke $\to$ gleiche CT (Pinguin-Angriff)}\\
\textbf{CTR} & $c_i=E(k,N_i)\xor m_i$, $N_i=N\|i$.
  Pad: nein. Streamcipher-artig, parallelisierbar, nur $E()$ nötig.
  $N$ = Nonce($n$\,Bit)$\|$Counter($r$\,Bit), $n{+}r=128$.
  Max.\ Blöcke pro $(N,K)$: $2^r$\\
\textbf{CBC} & $c_i=E(k,m_i\xor c_{i-1})$, $c_0{=}$IV.
  Pad: ja. Sequentiell. $D$: $m_i=D(k,c_i)\xor c_{i-1}$\\
\end{tabularx}
\tabend

\hh{IV/CT-Manipulation (Malleability)}

\tabstart
\begin{tabularx}{\linewidth}{K{0.13\linewidth}K{0.42\linewidth}L}
\THH{Modus} & \THH{Bit-Flip in CT} & \THH{IV-Änderung}\\
\textbf{CTR} & Flip $c_i$-Bit $\Rightarrow$ exakt selbes PT-Bit flippt (gezielte!) & ganzer Keystream korrupt\\
\textbf{CBC} & Flip $c_{i-1}$-Bit $\Rightarrow$ flippt $m_i$-Bit;
  $m_{i-1}$ komplett kaputt & nur 1.\ Block betroffen\\
\end{tabularx}
\tabend

\begin{wbox}
\textbf{CTR \& CBC: nur Vertraulichkeit, keine Integrität!}
Für aktiven Angreifer $\to$ AEAD!
\end{wbox}
\textbf{Vaudenay Padding Oracle:} Angreifer erfährt ob Padding falsch
$\to$ CBC-CT entschlüsselbar. Bsp: ASP.NET, \textbf{POODLE} (SSL~3.0).

\h{6 · Angreifermodelle}

\tabstart
\begin{tabularx}{\linewidth}{K{0.12\linewidth}L}
\THH{Modell} & \THH{Fähigkeit / Beispiel}\\
\textbf{EAV} & nur Challenge-CT -- ECB bereits hier unsicher\\
\textbf{KPA} & $(m,c)$-Paare bekannt -- ROT sofort gebrochen\\
\textbf{CPA} & Verschlüsselungs-Orakel -- Midway, Web-Embedding\\
\textbf{CCA} & Ver- \textbf{und} Entschlüsselungs-Orakel -- Router leitet CT\\
\end{tabularx}
\tabend

\begin{ibox}
\textbf{PRF:} Angreifer kann $F(k,\cdot)$ nicht von Zufallsfunktion unterscheiden.\\
AES-CTR/CBC: \good{CPA-sicher} (falls AES PRF), \warn{nicht CCA} (malleable)\\
AES-GCM: \good{CCA-sicher} (AEAD, Manipulation vor Entschlüsselung erkannt)
\end{ibox}

\h{7 · MAC \& Authenticated Encryption}
\begin{itemize}
\item \textbf{MAC} $M_k(m)$: Tag $T$ -- schützt \textbf{Integrität + Authentizität}
  (keine Vertraulichkeit!). Ziel: kein forgery under chosen-message.
  Konstruktionen: HMAC, Poly1305, CMAC
\end{itemize}
\begin{ibox}
\textbf{AE/AEAD:} $E(K,P)=(C,T)$; $D(K,C,T)=P$ oder \textbf{error}.
Erkennt Manipulation VOR Entschlüsselung $\to$ CCA-sicher.\\
\textbf{AES-GCM:} populärstes AEAD.
\textbf{ChaCha20-Poly1305:} TLS~1.3.\\
\textbf{Encrypt-then-MAC:} erst CT, dann MAC -- alles andere unsicher!
Nie selbst implementieren.\\
\end{ibox}

$\to$ H(salt$\|$pw), Salt zufällig pro Nutzer $\to$ Vorab-DB nutzlos\\

\h{10 · Zufallszahlen (RNG)}
\begin{itemize}
\item \warn{PRNG (nie für Krypto!):} java.util.Random,
  Python \cd{random()} -- vorhersagbar, rückrechenbar
\item \good{CSPRNG:} Linux: \cd{/dev/urandom}, \cd{getrandom()};
  Windows: \cd{BCryptGenRandom}; Intel rdrand (nur 1 Quelle)
\item \warn{Boot-Zeit = wenig Entropie!} HW-RNG nur eine Quelle von mehreren
\end{itemize}
\begin{ibox}
\textbf{Fortuna (FreeBSD):} Zustand $S=(C,K)$.
GenerateBlocks: $r\|{=}E_K(C);\; C\!{+\!+}$.
Key-Rekeying $K\leftarrow$GenerateBlocks$(S,2)$ nach Output.\\
\textbf{Backtracking:} aus $K_i$ nicht $K_{i-1}$ berechenbar
(Vergangenheit geheim)\\
\textbf{Forward Security:} neuer Zustand bei unbekanntem $K$ uniform
\end{ibox}

\h{11 · RSA}
\begin{itemize}
\item Setup: $p,q$ prim; $N=p\cdot q$;\;
  $e\cdot d\equiv 1\!\pmod{(p-1)(q-1)}$.
  Public $(e,N)$, Private $(d,N)$
\item Enc: $c=m^e\bmod N$.\quad Dec: $m=c^d\bmod N$.
  Korrektheit: $c^d=m^{ed}=m$ (Euler/Fermat)
\end{itemize}
\begin{ibox}
\textbf{Rechenrezept:} Square-and-Multiply --
\textbf{nach jedem Schritt $\bmod N$!}\\
$x\bmod N = x - \lfloor x/N\rfloor\cdot N$.\\
Bsp: $N{=}187$, $e{=}3$, $m{=}11$:
$11^3{=}1331 \bmod 187 = 1331-7{\cdot}187 = \mathbf{22}$\\
DH-Bsp: $N{=}17$, $g{=}3$, $a{=}3$:
$A=3^3{=}27 \bmod 17=\mathbf{10}$.
$B{=}9$: $K=9^3{=}729 \bmod 17=\mathbf{15}$
\end{ibox}
\begin{wbox}
Textbuch-RSA ohne Padding unsicher!
Praxis: RSA-OAEP (Verschlüsselung), RSA-PSS (Signatur)
\end{wbox}

\h{12 · Diffie-Hellman \& ECC}
\begin{itemize}
\item \textbf{DH:} $N$ prim, $g$ Generator.
  Alice: $a\leftarrow\mathrm{rand}$, schickt $A=g^a\!\bmod N$.
  Bob: $b\leftarrow\mathrm{rand}$, schickt $B=g^b\!\bmod N$.
  Gemeinsam: $K=B^a=A^b=g^{ab}\!\bmod N$
\item Sicherheit: Diskreter Logarithmus ($a$ aus $g^a\bmod N$) schwer
\end{itemize}
\begin{wbox}
\textbf{DH ohne Auth $\to$ MITM!} Eve macht 2 separate Austausche.
Fix: Signaturen + Zertifikate (ECDHE-Variante in TLS)
\end{wbox}
\begin{itemize}
\item \textbf{ECC:} $\approx$256\,Bit ECC $\approx$ RSA-3072.
  Schneller, kleinere Schlüssel
\item \textbf{Quanten/Shor:} bricht RSA, DH, ECC.
  Q-Day $\approx$2029. \textbf{Post-Quantum:}
  ML-KEM (Kyber, KEX), ML-DSA (Dilithium, Sig.)
\item \textbf{PFS:} ephemere (EC)DHE-Schlüssel nach Session gelöscht
  $\to$ aufgezeichneter Traffic nicht entschlüsselbar.
  ,,Harvest now, decrypt later''
\end{itemize}

\h{13 · Digitale Signaturen}
\begin{itemize}
\item $\mathrm{Sign}_{\mathrm{priv}}(m)=\sigma$;\;
  $\mathrm{Verify}_{\mathrm{pub}}(m,\sigma)=\mathrm{true/false}$
\item Gibt: \textbf{Authentizität + Integrität + Nicht-Abstreitbarkeit}
\item RSA-PSS, ECDSA. Sicher unter EUF-CMA.
  In Praxis: $\mathrm{Sign}(H(m))$ (für große Nachrichten)
\end{itemize}

\h{14 · PKI \& X.509-Zertifikate}
\begin{itemize}
\item \textbf{Zertifikat:} Subject (Identität) an Public Key gebunden,
  signiert vom Issuer.
  Felder: Subject, Issuer, Valid From/Until, PublicKey,
  Extensions (CA-Flag, pathlen), Signatur
\end{itemize}
\begin{tbox}
Kette: Blatt(Server) $\to$ Zwischen-CA $\to$ Root-CA\\
Root: Subject=Issuer (self-signed). Issuer[i] = Subject[i+1]
\end{tbox}
\textbf{Gültigkeitsprüfung -- alle 4 Bedingungen:}
\begin{enumerate}
\item \textbf{Signaturen} jedes Zertifikats verifizieren
  (mit Public Key des Issuers)
\item \textbf{Zeit:} Valid From $\le$ jetzt $\le$ Valid Until für
  \textbf{jedes} Zertifikat
\item \textbf{Constraints:} CA=true für Zwischen-CAs;
  pathlen=$k$ $\to$ max.\ $k$ CAs unterhalb
\item \textbf{Trust Anchor:} Root-CA im \textbf{Client Trust Store}
  (vorkonfiguriert!)
\end{enumerate}

\h{15 · Kerberos}
\begin{itemize}
\item \textbf{KDC} = AS (Auth.\ Server) + TGS (Ticket Granting Server).
  Kennt Langzeit-Schlüssel aller Nutzer/Dienste
\item Login $\to$ TGT vom AS $\to$ Service-Ticket vom TGS $\to$ Dienst
\end{itemize}
\begin{wbox}
\textbf{KDC kompromittiert:} kennt alle Langzeit-Schlüssel $\to$
kann beliebige Tickets ausstellen $\to$ jede Identität vortäuschen.
KDC = Single Point of Trust/Failure!
\end{wbox}

\h{17 · Netzwerksicherheit -- Grundangriffe}

\tabstart
\begin{tabularx}{\linewidth}{K{0.22\linewidth}L}
\THH{Angriff} & \THH{Beschreibung / Gegenmaßnahme}\\

$\to$ Forward Secrecy
\end{ibox}

\h{18 · TLS 1.2 -- Handshake}
\begin{tbox}
C$\to$S: ClientHello(Ver, Random\_C, SessionID,\\
\hspace*{7mm} CipherSuites[, SessionTicket])\\
S$\to$C: ServerHello(Ver, Random\_S, Suite)\\
S$\to$C: Certificate\\
S$\to$C: ServerKeyExchange(g,N,g\^{}a; Sig.)[bei DHE]\\
S$\to$C: ServerHelloDone\\
C$\to$S: ClientKeyExchange(g\^{}b) [PMS=g\^{}\{ab\}]\\
C$\to$S: ChangeCipherSpec · Finished\\
S$\to$C: ChangeCipherSpec · Finished\\
MS=PRF(PMS,"master secret",RC||RS) $\to$ 4 Session-Keys
\end{tbox}
\begin{itemize}
\item \textbf{Ciphersuite (1.2):} KEX-Auth-Cipher-MAC.
  Bsp: \texttt{ECDHE-RSA-AES256-GCM-SHA384}
\item \textbf{Session-Ticket:} im ClientHello in Extension
  \texttt{session\_ticket}. Akzeptiert wenn Server mit
  ServerHello + ChangeCipherSpec + Finished antwortet
  (kein Cert/SKE = abgekürzter Handshake)
\item \textbf{Gewählte Suite:} steht im \textbf{ServerHello}
\end{itemize}

\h{19 · TLS 1.3 -- Vergleich}

\tabstart
\begin{tabularx}{\linewidth}{K{0.28\linewidth}L}
\THH{Aspekt} & \THH{TLS 1.2 vs.\ TLS 1.3}\\
\textbf{Handshake-RTT}  & 1.2: 2 RTT, RSA oder (EC)DHE $|$
  1.3: 1 RTT, nur (EC)DHE (PFS immer)\\
\textbf{Ciphersuite}    & 1.2: KEX-Auth-Cipher-MAC $|$
  1.3: \texttt{TLS\_<AEAD>\_<HASH>} (2 Teile, nur AEAD)\\

\h{21 · Downgrade-Angriffe}
\begin{itemize}
\item Angreifer manipuliert ClientHello $\to$ streicht starke Suites
  $\to$ Server wählt schwache Suite
\item \textbf{POODLE:} SSL~3.0 Fallback erzwingen $\to$
  Padding Oracle auf CBC
\item \textbf{FREAK:} schwache RSA-Exportkeys erzwingen.
  \textbf{SMACK/SLOTH:} MD5 in TLS erzwingen
\item \textbf{Fix TLS~1.3:} CertificateVerify über Transcript
  $\to$ unbemerkte Modifikation unmöglich
\end{itemize}

\h{22 · Protokoll-Sicherheitsanalyse -- Rezept}
\begin{ibox}
\textbf{1.\ Schutzziele:} Vertraulichkeit, Integrität, Auth., Replay, PFS?\\
\textbf{2.\ Primitive:} CPA oder CCA? AEAD? Signatur? MAC über was?\\
\textbf{3.\ Angriffsvektoren:}
\begin{itemize}[topsep=0pt,itemsep=0pt,parsep=0pt]
\item Replay? $\to$ fehlt Freshness (Nonce im MAC gebunden?)
\item MITM? $\to$ fehlt Auth.\ des KEX?
\item Downgrade? $\to$ Transcript gebunden?
\item Malleability? $\to$ CTR/CBC ohne MAC?
\item Nonce-Reuse? $\to$ Wie wird Nonce gewählt?
\end{itemize}
\textbf{4.\ Fix:} AEAD statt CPA; Nonce für Freshness;
Sign für Auth.; Transcript-Hash gegen Downgrade
\end{ibox}
